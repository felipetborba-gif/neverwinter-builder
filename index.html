# ============================================ # MMO Status Table Generator # Author: Felipe # Description: # Gera uma tabela estilo MMO com 5 colunas, # 5 linhas de valores inteiros e soma final. # ============================================ import random from prettytable import PrettyTable # Título estilizado print("\n" + "="*55) print(" ⚔️ MMO CHARACTER STATS TABLE ⚔️") print("="*55 + "\n") # Colunas da tabela columns = ["Power", "Accuracy", "Combat Advantage", "Critical Strike", "Critical Severity"] # Criação da tabela table = PrettyTable() table.field_names = columns # Geração das 5 linhas com números inteiros rows = [] for _ in range(5): row = [random.randint(10, 999) for _ in range(len(columns))] rows.append(row) table.add_row(row) # Cálculo das somas sums = [sum(col) for col in zip(*rows)] # Linha final com somas table.add_row(["-"*10]*5) table.add_row(sums) # Estilo da tabela table.align = "r" print(table) print("\n" + "="*55) print(" Generated for MMO builds and theorycrafting") print("="*55 + "\n")
